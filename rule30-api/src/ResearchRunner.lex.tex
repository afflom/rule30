definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "The runner composes research attention with local process supervision. A work action requests a concrete strategy job; a review action requires the verified report of a successful process. Only that report may supply an inspected sequent or checked proof answer. Successful exit alone cannot establish a task. Changed dependencies cancel the active job and discard stale review evidence. A host protocol or cleanup failure halts the runner while preserving its outstanding request for recovery. The generated driver, strategy job construction, report decoding and durable publication remain separate execution-boundary responsibilities.",
  executable: true
}

definition{
  name: current,
  parameters: [research : Research::State, request : Research::Request],
  result: Bool,
  body: ((Research::sameSnapshot (project research snapshot) (project request snapshot)) && ((Research::componentReady (project (project request task) component) (project research components)) && ((Research::requestStillCurrent request (project research tasks)) && (match (project research pending) | Option.none => false | Option.some pending => (Research::sameRequest pending request))))),
  executable: true
}

definition{
  name: forward,
  parameters: [state : State, result : (Research::State * Research::Action)],
  result: (State * Action),
  body: ((record State {research := (first result), execution := (project state execution), review := (project state review), halted := (project state halted)}), (match (second result)
    | Research::Action.run request => (ctor Action.work request)
    | Research::Action.wait => (ctor Action.wait)
    | Research::Action.idle => (ctor Action.idle)
    | Research::Action.blocked => (ctor Action.blocked)
    | Research::Action.refused => (ctor Action.refused))),
  executable: true
}

definition{
  name: clear,
  parameters: [research : Research::State, halted : Bool],
  result: State,
  body: (record State {research := research, execution := (ctor Option.none [ResearchExecution::State]), review := (ctor Option.none [ResearchExecution::Completion]), halted := halted}),
  executable: true
}

definition{
  name: answer,
  parameters: [research : Research::State, request : Research::Request, result : Research::ResultKind],
  result: (State * Action),
  body: (forward (clear research false) (Research::receive research (record Research::Answer {request := request, result := result}))),
  executable: true
}

definition{
  name: completed,
  parameters: [research : Research::State, execution : ResearchExecution::State, completion : ResearchExecution::Completion],
  result: (State * Action),
  body: (let request : Research::Request := (project completion request); (if (not (ResearchExecution::resumeAllowed completion)) then ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), halted := true}), (ctor Action.blocked)) else (if (not (current research request)) then (answer research request (ctor Research::ResultKind.unavailable)) else (match (project completion kind)
    | ResearchExecution::EndKind.succeeded => ((record State {research := research, execution := (ctor Option.none [ResearchExecution::State]), review := (ctor Option.some [ResearchExecution::Completion] completion), halted := false}), (ctor Action.review completion))
    | ResearchExecution::EndKind.failed => (answer research request (ctor Research::ResultKind.failed (prim utf8_encode (project completion errors) : Bytes)))
    | ResearchExecution::EndKind.cut => (answer research request (ctor Research::ResultKind.cut))
    | ResearchExecution::EndKind.unavailable => (answer research request (ctor Research::ResultKind.unavailable))
    | ResearchExecution::EndKind.protocol => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), halted := true}), (ctor Action.blocked))
    | ResearchExecution::EndKind.cleanupFailed => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), halted := true}), (ctor Action.blocked))
    | ResearchExecution::EndKind.refused => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), halted := true}), (ctor Action.blocked)))))),
  executable: true
}

definition{
  name: executionResult,
  parameters: [research : Research::State, result : (ResearchExecution::State * ResearchExecution::Emission)],
  result: (State * Action),
  body: (match (second result)
    | ResearchExecution::Emission.complete completion => (completed research (first result) completion)
    | ResearchExecution::Emission.awaiting => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] (first result)), review := (ctor Option.none [ResearchExecution::Completion]), halted := false}), (ctor Action.wait))
    | ResearchExecution::Emission.host request => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] (first result)), review := (ctor Option.none [ResearchExecution::Completion]), halted := false}), (ctor Action.host request))),
  executable: true
}

definition{
  name: begin,
  parameters: [research : Research::State],
  result: (State * Action),
  body: (match (project research pending)
    | Option.some pending => ((clear research true), (ctor Action.blocked))
    | Option.none => (forward (clear research false) (Research::dispatch research))),
  axioms: [],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}

definition{
  name: startJob,
  parameters: [state : State, job : ResearchExecution::Job],
  result: (State * Action),
  body: (match (project state execution)
    | Option.some execution => (state, (ctor Action.refused))
    | Option.none => (match (project state review)
      | Option.some completion => (state, (ctor Action.refused))
      | Option.none => (if ((current (project state research) (project job request)) && (ResearchExecution::validJob job)) then (executionResult (project state research) (ResearchExecution::begin job)) else (state, (ctor Action.refused))))),
  executable: true
}

definition{
  name: reviewed,
  parameters: [state : State, receipt : Research::Answer],
  result: (State * Action),
  body: (match (project state review)
    | Option.none => (state, (ctor Action.refused))
    | Option.some completion => (if ((Research::sameRequest (project completion request) (project receipt request)) && (current (project state research) (project receipt request))) then (forward (clear (project state research) false) (Research::receive (project state research) receipt)) else (state, (ctor Action.refused)))),
  executable: true
}

definition{
  name: reconcile,
  parameters: [state : State, result : (Research::State * Research::Action)],
  result: (State * Action),
  body: (let research : Research::State := (first result); (match (project state execution)
    | Option.some execution => (if (current research (project (project execution job) request)) then ((record State {research := research, execution := (project state execution), review := (project state review), halted := false}), (ctor Action.wait)) else (executionResult research (ResearchExecution::cancel execution)))
    | Option.none => (match (project state review)
      | Option.some completion => (if (current research (project completion request)) then ((record State {research := research, execution := (project state execution), review := (project state review), halted := false}), (ctor Action.review completion)) else (answer research (project completion request) (ctor Research::ResultKind.unavailable)))
      | Option.none => (forward state result)))),
  executable: true
}

definition{
  name: step,
  parameters: [state : State, event : Event],
  result: (State * Action),
  body: (if (project state halted) then (state, (ctor Action.blocked)) else (match event
    | Event.job job => (startJob state job)
    | Event.reviewed receipt => (reviewed state receipt)
    | Event.changed snapshot components affected => (reconcile state (Research::step (project state research) (ctor Research::Event.changed snapshot components affected)))
    | Event.ingest hypotheses => (reconcile state (Research::step (project state research) (ctor Research::Event.ingest hypotheses)))
    | Event.tick => (match (project state execution)
      | Option.some execution => (state, (ctor Action.wait))
      | Option.none => (match (project state review)
        | Option.some completion => (state, (ctor Action.review completion))
        | Option.none => (forward state (Research::dispatch (project state research)))))
    | Event.host hostEvent => (match (project state execution)
      | Option.none => (state, (ctor Action.refused))
      | Option.some execution => (executionResult (project state research) (ResearchExecution::step execution hostEvent)))
    | Event.cancel => (match (project state execution)
      | Option.some execution => (executionResult (project state research) (ResearchExecution::cancel execution))
      | Option.none => (match (project state review)
        | Option.some completion => (answer (project state research) (project completion request) (ctor Research::ResultKind.cut))
        | Option.none => (state, (ctor Action.refused)))))),
  axioms: [],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}

theorem{
  name: haltedRunnerNeverDispatches,
  parameters: [research : Research::State, execution : (Option ResearchExecution::State), review : (Option ResearchExecution::Completion), event : Event],
  statement: ((step (record State {research := research, execution := execution, review := review, halted := true}) event) = ((record State {research := research, execution := execution, review := review, halted := true}), (ctor Action.blocked))),
  proof: reflexivity{}
}
