definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "The runner composes research attention with local process supervision. A work action requests a concrete strategy job; a review action requires the verified report of a successful process. Inspection and maintenance reports remain distinct from proof verification. Event.candidate owns the local compiler snapshot, derives candidate identities, checks the contextual type and transfers ownership to proof verification. A checked proof answer is produced only by the owned ResearchProof verifier after its report is bound to the exact candidate type; the reviewed event refuses externally supplied checked answers. Successful exit alone cannot establish a task. Changed dependencies cancel the active job and discard stale review evidence. A host protocol or cleanup failure halts the runner while preserving its outstanding request for recovery. The generated driver, strategy job construction, source inspection and durable publication remain separate execution-boundary responsibilities.",

definition{
  name: completed,
  parameters: [research : Research::State, execution : ResearchExecution::State, completion : ResearchExecution::Completion],
  result: (State * Action),
  body: (let request : Research::Request := (project completion request); (if (not (ResearchExecution::resumeAllowed completion)) then ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), proof := (ctor Option.none [ResearchProof::State]), candidate := (ctor Option.none [ResearchCandidate::State]), halted := true}), (ctor Action.blocked)) else (if (not (current research request)) then (answer research request (ctor Research::ResultKind.unavailable)) else (match (project completion kind)
    | ResearchExecution::EndKind.succeeded => ((record State {research := research, execution := (ctor Option.none [ResearchExecution::State]), review := (ctor Option.some [ResearchExecution::Completion] completion), proof := (ctor Option.none [ResearchProof::State]), candidate := (ctor Option.none [ResearchCandidate::State]), halted := false}), (ctor Action.review completion))
    | ResearchExecution::EndKind.failed => (answer research request (ctor Research::ResultKind.failed (prim utf8_encode (project completion errors) : Bytes)))
    | ResearchExecution::EndKind.cut => (answer research request (ctor Research::ResultKind.cut))
    | ResearchExecution::EndKind.unavailable => (answer research request (ctor Research::ResultKind.unavailable))
    | ResearchExecution::EndKind.protocol => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), proof := (ctor Option.none [ResearchProof::State]), candidate := (ctor Option.none [ResearchCandidate::State]), halted := true}), (ctor Action.blocked))
    | ResearchExecution::EndKind.cleanupFailed => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), proof := (ctor Option.none [ResearchProof::State]), candidate := (ctor Option.none [ResearchCandidate::State]), halted := true}), (ctor Action.blocked))
    | ResearchExecution::EndKind.refused => ((record State {research := research, execution := (ctor Option.some [ResearchExecution::State] execution), review := (ctor Option.none [ResearchExecution::Completion]), proof := (ctor Option.none [ResearchProof::State]), candidate := (ctor Option.none [ResearchCandidate::State]), halted := true}), (ctor Action.blocked)))))),
  executable: true
}

definition{
  name: startJob,
  parameters: [state : State, job : ResearchExecution::Job],
  result: (State * Action),
  body: (match (project state execution)
    | Option.some execution => (state, (ctor Action.refused))
    | Option.none => (match (project state review)
      | Option.some completion => (state, (ctor Action.refused))
      | Option.none => (if ((current (project state research) (project job request)) && (ResearchExecution::validJob job)) then (executionResult (project state research) (ResearchExecution::begin job)) else (state, (ctor Action.refused))))),
  executable: true

    | Option.some completion => (if ((Research::sameRequest (project completion request) (project receipt request)) && (current (project state research) (project receipt request))) then (forward (clear (project state research) false) (Research::receive (project state research) receipt)) else (state, (ctor Action.refused)))),

definition{
  name: step,
  parameters: [state : State, event : Event],
  result: (State * Action),
  body: (if (project state halted) then (state, (ctor Action.blocked)) else (match (project state candidate)
    | Option.some candidate => (match event
      | Event.host response => (candidateResult (project state research) (ResearchCandidate::step candidate (ctor ResearchVerification::Event.host response)))
      | Event.cancel => (candidateResult (project state research) (ResearchCandidate::step candidate (ctor ResearchVerification::Event.cancel)))
      | Event.changed snapshot components affected => (reconcile state (Research::step (project state research) (ctor Research::Event.changed snapshot components affected)))
      | Event.ingest hypotheses => (reconcile state (Research::step (project state research) (ctor Research::Event.ingest hypotheses)))
      | Event.tick => (state, (ctor Action.wait))
      | Event.job job => (state, (ctor Action.refused))
      | Event.proof pins config binding maxDepth maxSteps => (state, (ctor Action.refused))
      | Event.candidate context => (state, (ctor Action.refused))
      | Event.reviewed receipt => (state, (ctor Action.refused)))
    | Option.none => (match (project state proof)
    | Option.some proof => (match event
      | Event.host response => (proofResult (project state research) (ResearchProof::step proof (ctor ResearchVerification::Event.host response)))
      | Event.cancel => (proofResult (project state research) (ResearchProof::step proof (ctor ResearchVerification::Event.cancel)))
      | Event.changed snapshot components affected => (reconcile state (Research::step (project state research) (ctor Research::Event.changed snapshot components affected)))
      | Event.ingest hypotheses => (reconcile state (Research::step (project state research) (ctor Research::Event.ingest hypotheses)))
      | Event.tick => (state, (ctor Action.wait))
      | Event.job job => (state, (ctor Action.refused))
      | Event.proof pins config binding maxDepth maxSteps => (state, (ctor Action.refused))
      | Event.candidate context => (state, (ctor Action.refused))
      | Event.reviewed receipt => (state, (ctor Action.refused)))
    | Option.none => (match event
    | Event.proof pins config binding maxDepth maxSteps => (startProof state pins config binding maxDepth maxSteps)
    | Event.candidate context => (startCandidate state context)
    | Event.job job => (startJob state job)
    | Event.reviewed receipt => (reviewed state receipt)
    | Event.changed snapshot components affected => (reconcile state (Research::step (project state research) (ctor Research::Event.changed snapshot components affected)))
    | Event.ingest hypotheses => (reconcile state (Research::step (project state research) (ctor Research::Event.ingest hypotheses)))
    | Event.tick => (match (project state execution)
      | Option.some execution => (state, (ctor Action.wait))
      | Option.none => (match (project state review)
        | Option.some completion => (state, (ctor Action.review completion))
        | Option.none => (forward state (Research::dispatch (project state research)))))
    | Event.host hostEvent => (match (project state execution)
      | Option.none => (state, (ctor Action.refused))
      | Option.some execution => (executionResult (project state research) (ResearchExecution::step execution hostEvent)))
    | Event.cancel => (match (project state execution)
      | Option.some execution => (executionResult (project state research) (ResearchExecution::cancel execution))
      | Option.none => (match (project state review)
        | Option.some completion => (answer (project state research) (project completion request) (ctor Research::ResultKind.cut))
        | Option.none => (state, (ctor Action.refused)))))))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, files, overflow, process, recursion], targets: [rust-std]}
}
