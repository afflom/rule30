definition{
  name: reading,
  parameters: [state : State, event : LexLeanStd.Host::Event],
  result: (State * Emission),
  body: (match (readResponse event)
 | Option.none => (finish state (ctor Outcome.rejected "unexpected verification host response"))
 | Option.some response => (match (project state cancelled) | Bool.true => (finish state (ctor Outcome.cancelled)) | Bool.false => (match response
   | Result.error problem => (finish state (ctor Outcome.rejected "verification report read failed"))
   | Result.ok source => (match (project state phase)
     | Phase.command => (commandRead state source)
     | Phase.attestation command => (attestationRead state command source)
     | Phase.running => (finish state (ctor Outcome.rejected "no pending verification read"))
     | Phase.ended => (state, (ctor Emission.awaiting)))))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: step,
  parameters: [state : State, event : Event],
  result: (State * Emission),
  body: (match (project state phase)
 | Phase.ended => (state, (ctor Emission.awaiting))
 | Phase.running => (match event
   | Event.cancel => (cancel state)
   | Event.changed snapshot => (if (Research::sameSnapshot snapshot (project (project (project state pins) request) snapshot)) then (state, (ctor Emission.awaiting)) else (cancel state))
   | Event.host response => (executionResult state (ResearchExecution::step (project state execution) response)))
 | Phase.command => (match event
   | Event.cancel => (cancel state)
   | Event.changed snapshot => (if (Research::sameSnapshot snapshot (project (project (project state pins) request) snapshot)) then (state, (ctor Emission.awaiting)) else (cancel state))
   | Event.host response => (reading state response))
 | Phase.attestation command => (match event
   | Event.cancel => (cancel state)
   | Event.changed snapshot => (if (Research::sameSnapshot snapshot (project (project (project state pins) request) snapshot)) then (state, (ctor Emission.awaiting)) else (cancel state))
   | Event.host response => (reading state response))),
