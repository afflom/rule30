definition{
 name: contract,
 parameters: [],
 result: String,
 body: "Run at most 64 distinct candidate attempts sequentially against one owned pinned library. Admission requires identical library configuration, tool identities and snapshots. Each candidate retains its own request, source directory, verification and evidence record. A normally completed failed candidate permits the next attempt; a blocked or interrupted attempt stops the cohort and retains the library for recovery. Cancellation drains the outstanding host reply before library release. Only the cohort releases the shared library, after all active candidate work has completed. Completion reports per-attempt outcomes and does not imply that every candidate was proved. This bounded executor consumes supplied candidates; it does not generate strategies or provide durable restart recovery.",

 executable: true

definition{
 name: host,
 parameters: [state : State, event : LexLeanStd.Host::Event],
 result: (State * Emission),
 body: (match (project state phase)
 | Phase.preparing => (match (project state owner) | Option.none => (blocked state "missing preparing owner") | Option.some owner => (ownerResult state (ResearchWorkspace::step owner (ctor ResearchWorkspace::Event.host event))))
 | Phase.releasing => (match (project state owner) | Option.none => (blocked state "missing releasing owner") | Option.some owner => (ownerResult state (ResearchWorkspace::step owner (ctor ResearchWorkspace::Event.host event))))
 | Phase.attempting => (match (project state active) | Option.none => (blocked state "missing candidate owner") | Option.some active => (attemptResult state (ResearchSession::step active (ctor ResearchSession::Event.host event))))
 | Phase.advancing => (match (ResearchExecution::asWaited event) | Option.none => (blocked state "unexpected cohort continuation reply") | Option.some outcome => (match outcome | Result.error failure => (blocked state "cohort continuation unavailable") | Result.ok reply => (match (project reply finished) | List.nil => (advance state) | List.cons head tail => (blocked state "unowned child in cohort continuation"))))
 | Phase.ended => (finish state)
 | Phase.blocked => (blocked state "cohort retained for recovery")),
 axioms: [Quot.sound, propext],
 executable: true
}
definition{
 name: cancel,
 parameters: [state : State],
 result: (State * Emission),
 body: (let next : State := (markCancelled state); (match (project state phase)
 | Phase.preparing => (match (project state owner) | Option.none => (blocked next "missing preparing owner") | Option.some owner => (ownerResult next (ResearchWorkspace::step owner (ctor ResearchWorkspace::Event.cancel))))
 | Phase.attempting => (match (project state active) | Option.none => (blocked next "missing candidate owner") | Option.some active => (attemptResult next (ResearchSession::step active (ctor ResearchSession::Event.cancel))))
 | Phase.advancing => (next, (ctor Emission.awaiting))
 | Phase.releasing => (next, (ctor Emission.awaiting))
 | Phase.ended => (finish state)
 | Phase.blocked => (blocked next "cohort retained for recovery"))),
 axioms: [Quot.sound, propext],
 executable: true
}
