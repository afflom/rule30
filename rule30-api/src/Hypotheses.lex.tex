definition{
  name: purpose,
  parameters: [],
  result: String,
  body: "The engine's role is to label new hypotheses. A hypothesis is a statement minted from data the engine already keeps, in a canonical form, with its sources: the exclusion a failed library-rewrite attempt names (for all t, not these atoms with these values together), or an open hypothesis of a conditional refutation in the audit. Its label is the SHA-256 of the canonical bytes of the statement, with no prefix or framing added to those bytes, as UOR-Foundation/uor-addr mints labels. An exclusion is a consequence of the relation whose attempt named it, never evidence for it: the attempt found an assignment that breaks the relation, so wherever the relation holds that assignment does not occur. This module proves none of the hypotheses and enumerates no cell or time.",

definition{
  name: lessFrom,
  parameters: [left : Bytes, right : Bytes, position : Nat, fuel : Nat],
  result: Bool,
  body: (match fuel
    | Nat.zero => false
    | Nat.succ more => (match (prim index left position : (Option UInt8))
      | Option.none => (match (prim index right position : (Option UInt8))
        | Option.some y => true
        | Option.none => false)
      | Option.some x => (match (prim index right position : (Option UInt8))
        | Option.none => false
        | Option.some y => (if (prim less_than x y : Bool) then true else (if (prim equal x y : Bool) then (lessFrom left right (position + 1) more) else false))))),
  axioms: [propext],
  recursive_argument: fuel,
  executable: true
}
